\begin{frame}{Master Algorithms}
  \centering
  {\small\itshape Pluggable algorithm interface --- System and
   time-advancement strategy are separated.}
  \vspace{0.8em}

  \footnotesize
  \begin{tabular}{
      l
      >{\raggedright\arraybackslash}p{0.42\textwidth}
      >{\raggedright\arraybackslash}p{0.30\textwidth}}
    \toprule
    \textbf{Algorithm} & \textbf{Strategy} & \textbf{Use when} \\
    \midrule

    Jacobi &
    Prepare all interfaces $\to$ step all components. \newline
    \ac{SCC}-local solver for loops. &
    Loose coupling; \newline parallelism matters \\
    \addlinespace[3pt]

    Gauss--Seidel &
    Generation-wise: prepare $\to$ step per generation. \newline
    \ac{SCC}-local solver for loops. &
    \textbf{Default continuous}; \newline tighter coupling \\
    \addlinespace[3pt]

    \ac{IJCSA} &
    \textbf{Global} interface Newton solve --- one coupled residual
    for all zero-delay inputs $\to$ step all. &
    Many or strongly-coupled loops across the system \\
    \midrule

    Hybrid &
    Wraps a continuous algorithm; adds event detect / localize /
    dispatch. &
    \emph{Auto-selected} when any component declares an event indicator \\
    \bottomrule
  \end{tabular}

  \vspace{1em}
  \footnotesize
  \emph{Hybrid is automatically activated by the framework;
        the user only chooses the continuous algorithm.}
\end{frame}

\begin{document}

% Parameter definitions
\def\boxW{3}
\def\boxH{2}

\begin{tikzpicture}[
    component/.style={
        draw, rectangle, minimum width=2.6cm, minimum height=1.6cm,
        fill=white, thick, align=center
    },
    conn/.style={-latex, thick},
]

% Draw grid for reference
% \draw[dashed, step=0.5cm, help lines ] (-7,-3) grid (7,3);

% Subsystem A
\begin{scope}[shift={(-\boxW,0)}]
    \draw[thick, fill=blue!20] (-\boxW/2, -\boxH/2) rectangle (\boxW/2, \boxH/2);
    \node at (0,\boxH/3) {Subsystem $\mathrm{A}$\\[2pt]};
    \node at (0,0) {\footnotesize $\dot x_\mathrm{A}\!=\!f_\mathrm{A}(x_\mathrm{A},u_\mathrm{A}, t)$};
    \node at (0, -\boxH/4) {\footnotesize $y_\mathrm{A}\!=\!h_\mathrm{A}(x_\mathrm{A},u_\mathrm{A})$};
\end{scope}

% Subsystem B
\begin{scope}[shift={(\boxW,0)}]
    \draw[thick, fill=orange!20] (-\boxW/2, -\boxH/2) rectangle (\boxW/2, \boxH/2);
    \node at (0,\boxH/3) {Subsystem $\mathrm{B}$\\[2pt]};
    \node at (0,0) {\footnotesize $\dot x_\mathrm{B}\!=\!f_\mathrm{B}(x_\mathrm{B},u_\mathrm{B}, t)$};
    \node at (0, -\boxH/4) {\footnotesize $y_\mathrm{B}\!=\!h_\mathrm{B}(x_\mathrm{B},u_\mathrm{B})$};
\end{scope}

% Connection A -> B
\draw[thick, ->] (-\boxW/2,0) -- node[above] {\footnotesize $u_\mathrm{B} = y_\mathrm{A}$} (\boxW/2,0);

% Connection B -> A
\draw[thick, ->] (\boxW/2 + \boxW, 0)
    --++ (0.5, 0)
    --++ (0, -1.5)
    -- node[midway, above] {\footnotesize $u_\mathrm{A} = y_\mathrm{B}$} ++(-3*\boxW -1, 0)
    --++ (0, 1.5)
    -- (-\boxW/2 - \boxW, 0);

\end{tikzpicture}

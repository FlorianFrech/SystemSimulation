% =============================================================================
% PID Controller Component
% =============================================================================
\newcommand{\PIDControllerComponent}{
  \def\xErr{-1}   % error summation x
  \def\xPID{3.5}   % PID branch boxes x
  \def\xCtrl{7}   % control summation x
  \def\wPID{3.5}
  \def\hPID{1.25}
  \def\rSum{0.5}

  \colorlet{sumColor}{green!90!black}
  \colorlet{pColor}{yellow!90!black}
  \colorlet{iColor}{cyan!90!black}
  \colorlet{dColor}{orange!90!black}

  \draw[thick, fill=white!95!black] (-3.5,-3.5) rectangle (9.5,4);
  % \draw [help lines, step=0.5] (-7, -7) grid (12,7);
  \node[font=\large\bfseries] at (3.25, 3.25) {PID Controller};

  \coordinate (SumE) at (\xErr,0);
  \coordinate (P) at (\xPID,2.0);
  \coordinate (I) at (\xPID,0.0);
  \coordinate (D) at (\xPID,-2.0);
  \coordinate (SumU) at (\xCtrl,0);
  \coordinate (RefIn) at (\xErr-2,0);
  \coordinate (MeasIn) at (\xErr, -3.5);
  \coordinate (ResetIn) at (0.5, -3.5);
  \coordinate (CtrlOut) at (\xCtrl+1.5,-3.5);

  % --- summation circles ---
  \draw[thick, fill=sumColor] (SumE) circle (\rSum);
  \node[font=\large\bfseries] at (SumE) {$\sum$};
  \draw[thick, fill=sumColor] (SumU) circle (\rSum);
  \node[font=\large\bfseries] at (SumU) {$\sum$};

  % --- P/I/D blocks ---
  \draw[thick, fill=pColor] ($(P)+(-\wPID/2,-\hPID/2)$) rectangle ($(P)+(\wPID/2,\hPID/2)$);
  \draw[thick, fill=iColor] ($(I)+(-\wPID/2,-\hPID/2)$) rectangle ($(I)+(\wPID/2,\hPID/2)$);
  \draw[thick, fill=dColor] ($(D)+(-\wPID/2,-\hPID/2)$) rectangle ($(D)+(\wPID/2,\hPID/2)$);
  \node[font=\large\bfseries, anchor=west] at ($(P)+(-\wPID/2+0.2,0)$) {P};
  \node[font=\large\bfseries, anchor=west] at ($(I)+(-\wPID/2+0.2,0)$) {I};
  \node[font=\large\bfseries, anchor=west] at ($(D)+(-\wPID/2+0.2,0)$) {D};
  \node[font=\large] at ($(P)+(0.3,0)$) {$K_p\,e(t)$};
  \node[font=\large] at ($(I)+(0.3,0)$) {$K_i\int_0^t e(\tau)\,\mathrm{d}\tau$};
  \node[font=\large] at ($(D)+(+0.3,0)$) {$K_d\,\dfrac{\mathrm{d}\theta_\mathrm{meas}(t)}{\mathrm{d}t}$};

  % --- reference, error, and branch split ---
  \draw[->, ultra thick] (-3.5,2) --++ (2.5,0) -- ($(SumE)+(0, \rSum)$) node[midway,above=1.25cm, left,font=\Large] {$\theta_\mathrm{ref}$};
  
  \draw[->, ultra thick] (MeasIn) -- ($(SumE)+(0,-\rSum)$) node[font=\Large] at (\xErr-1, -2){$\theta_\mathrm{meas}$};
  \node[font=\large\bfseries] at ($(SumE)+(-0.7,0.6)$) {$+$};
  \node[font=\large\bfseries] at ($(SumE)+(-0.35,-0.95)$) {$-$};

  \draw[->, ultra thick] ($(SumE)+(\rSum,0)$) -- ($(I)+(-\wPID/2,0)$) node[font=\Large] at (\xErr+1, 0.5) {$e$};
  \draw[ultra thick] ($(I)+(-\wPID/2,0)$) -- ($(I)+(-\wPID/2-0.9,0)$) -- ($(P)+(-\wPID/2-0.9,0)$) -- ($(P)+(-\wPID/2,0)$);
  \fill[black] ($(I)+(-\wPID/2-0.9,0)$) circle (0.1);
  \draw[->, ultra thick] ($(P)+(-\wPID/2-0.02,0)$) -- ($(P)+(-\wPID/2,0)$);

  % D-branch is derivative on measurement (not on error)
  \draw[->, ultra thick] (\xErr, -2) -- ($(D)+(-\wPID/2,0)$);
  \fill[black] (\xErr,-2) circle (0.1);

  % --- sum of P/I/D contributions ---
  \draw[->, ultra thick] ($(P)+(\wPID/2,0)$) -| ($(SumU)+(0,\rSum)$);
  \draw[->, ultra thick] ($(I)+(\wPID/2,0)$) -- ($(SumU)+(-\rSum,0)$);
  \draw[->, ultra thick] ($(D)+(\wPID/2,0)$) -| ($(SumU)+(0,-\rSum)$);
  \node[font=\large\bfseries] at ($(SumU)+(-0.86,0.35)$) {$+$};
  \node[font=\large\bfseries] at ($(SumU)+(-0.35,-0.95)$) {$+$};
  \node[font=\large\bfseries] at ($(SumU)+(-0.35,0.95)$) {$+$};

  % Optional integrator reset integrator
  \draw[->, ultra thick, dashed, red] (ResetIn) node[midway, below=1.0cm, right=0.5cm , font=\Large] {reset} -- (0.5, -0.4) -- (\xPID-\wPID/2, -0.4);

  % --- controller output and plant output ---
  \draw[->, ultra thick] ($(SumU)+(\rSum,0)$) -- ($(SumU)+(\rSum+1,0)$) -- (CtrlOut) node[above=0.75cm, left,font=\Large] {$u_\mathrm{control}$};

}

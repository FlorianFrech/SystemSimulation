% =============================================================================
% ADC Component
% =============================================================================
\newcommand{\ADCComponent}{
  \def\bw{5}         % component width
  \def\bh{4.5}         % component height
  \def\gW{3.6}       % mini-graph width
  \def\gH{2}       % mini-graph height
  \def\qStep{0.35}   % quantization step
  \def\Ts{0.45}      % sampling period

  \coordinate (C) at (0,0);

  \tikzset{adc-default/.style={line cap=round, line join=round}}

  % Component panel
  \draw[thick, fill=white!95!black] ($(C)+(-\bw/2,-\bh/2)$) rectangle ($(C)+(\bw/2,\bh/2)$);
  \node[font=\large\bfseries] at ($(C)+(0,\bh/2-0.55)$) {ADC};

  % Mini-graph: sample + quantization
  \begin{scope}[shift={(-\gW/2,-\gH/2+0.3)}]
    \draw[adc-default,step=\Ts,color=black!30] (0,0) grid (\gW,\gH);
    
    \draw[adc-default,->,thick] (0,0) -- (\gW+0.3,0) node[right] {$t$};
    \draw[adc-default,->,thick] (0,0) -- (0,\gH+0.3) node[above] {$V(t)$};

    \draw[adc-default,black!50,semithick,smooth,domain=0:\gW,samples=100]
      plot (\x,{1.1 + 0.55*sin(120*\x) + 0.25*sin(250*\x)});

    % Stair output with matched quantized sample values
    \foreach \i in {0,1,...,7} {
      \pgfmathsetmacro{\xs}{\i*\Ts}
      \pgfmathsetmacro{\xe}{(\i+1)*\Ts}
      \pgfmathsetmacro{\ys}{1.1 + 0.55*sin(120*\xs) + 0.25*sin(250*\xs)}
      \pgfmathsetmacro{\yq}{round(\ys/\qStep)*\qStep}
      \draw[adc-default,signalOutputColor,semithick] (\xs,\yq) -- (\xe,\yq);
      \fill[signalOutputColor] (\xs,\yq) circle(0.04);
    }
  \end{scope}

  \node[below, text width=4.2cm, align=center, information text] at ($(C)+(0,-1)$)
    {$\mathrm{nBits}$ (resolution)\\$f_s = 1/T_s$ (sampling rate)};
}

% =============================================================================
% Angle Sensor Component (Potentiometer + ADC)
% =============================================================================
\newcommand{\AngleSensorComponent}{
  \def\asW{5.5}         % outer box width
  \def\asH{13}        % outer box height
  \def\gW{4.1}        % mini-graph width
  \def\gH{2.9}          % mini-graph height
  \def\qStep{0.35}    % quantization step
  \def\Ts{0.45}       % sampling period

  \tikzset{as-default/.style={line cap=round, line join=round}}

  % --- Outer box ---
  \draw[thick, fill=white!95!black] (-\asW/2,-\asH/2) rectangle (\asW/2,\asH/2);
  \node[font=\large\bfseries] at (0,\asH/2-0.55) {Angle Sensor};

  % =========================================================================
  % Potentiometer (lower half)
  % =========================================================================
  \begin{scope}[shift={(0,-3.5)}]
    \pgfmathsetmacro{\potR}{2}
    \pgfmathsetmacro{\potTrackR}{0.67*\potR}
    \pgfmathsetmacro{\potWiperR}{0.25*\potR}
    \pgfmathsetmacro{\potRefR}{1.2*\potR}
    \pgfmathsetmacro{\potWAng}{45-90}
    \pgfmathsetmacro{\potTLW}{14*(\potR/2.0)}
    \pgfmathsetmacro{\potWLW}{7*(\potR/2.0)}
    \pgfmathsetmacro{\potTermLW}{7*(\potR/2.0)}

    % Circle body
    \coordinate (as-center) at (0,0);
    \draw[thick] (0,0) circle (\potR);

    % Resistive track
    \draw[line width=\potTLW pt, sensorTrackColor]
      (-225:\potTrackR) arc (-225:45:\potTrackR);

    % Wiper ring + arm
    \draw[line width=\potWLW pt, signalOutputColor] (0,0) circle (\potWiperR);
    \draw[line width=\potWLW pt, signalOutputColor]
      (\potWAng:\potWiperR) -- (\potWAng:\potTrackR);

    % V+, GND terminals
    \coordinate (as-gnd0) at (-225:\potTrackR-0.25);
    \coordinate (as-gnd1) at ($(-225:\potTrackR-0.15)+(-225:\potR*0.68)$);
    \coordinate (as-vp0) at (45:\potTrackR-0.25);
    \coordinate (as-vp1) at ($(45:\potTrackR-0.15)+(45:\potR*0.68)$);
    \draw[line width=\potTermLW pt, groundTerminalColor] (as-gnd0) -- (as-gnd1)
      node[above, font=\footnotesize] {GND};
    \draw[line width=\potTermLW pt, supplyTerminalColor] (as-vp0) -- (as-vp1)
      node[above, font=\footnotesize] {$V_+$};

    % Vout terminal (extends toward ADC graph)
    \draw[line width=\potTermLW pt, signalOutputColor]
      (90:\potWiperR) -- (90:\potR+0.35)
      node[right, font=\footnotesize, signalOutputColor] {$V_\mathrm{pot}(\theta)$};

    % Reference lines + θ arc
    \coordinate (as-down) at (0,-\potRefR);
    \coordinate (as-wray) at (\potWAng:\potRefR);
    \draw[dash dot, thick, black] (-\potRefR,0) -- (\potRefR,0);
    \draw[dash dot, thick, black] (0,-\potRefR) -- (0,\potRefR);
    \draw[dash dot, thick, black] (0,0) -- (as-wray);
    \draw[<->] (40:{0.44*\potR}) arc (40:-220:{0.44*\potR});
    \draw pic[->,"\,$\theta$", draw,
      angle radius=\potRefR cm, angle eccentricity=1.15]
      {angle=as-down--as-center--as-wray};

    % Coordinate system at the potentiometer/rotation axis.
    \pic at (as-center) {coordsys};

    % Sub-label
    \node[font=\footnotesize\bfseries] at (0,\potR+1) {Potentiometer};
  \end{scope}

  % =========================================================================
  % Internal connection: V_pot → ADC
  % =========================================================================
  \draw[line width=3.5pt, ->, signalOutputColor] (0,-0.15) --
    node[right=5pt, font=\footnotesize, signalOutputColor] {$V_\mathrm{pot}(\theta)$} (0,1.5);

  % =========================================================================
  % ADC mini-graph (upper half)
  % =========================================================================
  \node[font=\footnotesize\bfseries] at (0,5) {ADC};

  \begin{scope}[shift={(-\gW/2, 3-\gH/2)}]
    \draw[as-default,step=\Ts,color=black!30] (0,0) grid (\gW,\gH);

    \draw[as-default,->,thick] (0,0) -- (\gW+0.3,0) node[right, font=\scriptsize] {$t$};
    \draw[as-default,->,thick] (0,0) -- (0,\gH+0.3) node[above, font=\scriptsize] {$V$};

    % Continuous signal: V_pot(t)
    \draw[as-default,signalOutputColor,semithick,smooth,domain=0:\gW,samples=100]
      plot (\x,{1.1 + 0.55*sin(120*\x) + 0.25*sin(250*\x)});

    % Staircase: V_adc(t) — sample-and-hold output
    \foreach \i in {0,1,...,7} {
      \pgfmathsetmacro{\xs}{\i*\Ts}
      \pgfmathsetmacro{\xe}{(\i+1)*\Ts}
      \pgfmathsetmacro{\ys}{1.1 + 0.55*sin(120*\xs) + 0.25*sin(250*\xs)}
      \pgfmathsetmacro{\yq}{round(\ys/\qStep)*\qStep}
      \draw[as-default,orange!85!black,semithick] (\xs,\yq) -- (\xe,\yq);
      \fill[orange!85!black] (\xs,\yq) circle(0.04);
    }

    % Legend
    \fill[white, opacity=0.85] (0.05, \gH-0.7) rectangle (2, \gH+0.02);
    \draw[signalOutputColor, semithick] (0.15, \gH-0.15) -- ++(0.3,0);
    \node[right, font=\small] at (0.45, \gH-0.15) {$V_\mathrm{pot}(t)$};
    \draw[orange!85!black, semithick] (0.15, \gH-0.5) -- ++(0.3,0);
    \node[right, font=\small] at (0.45, \gH-0.5) {$V_\mathrm{adc}(t)$};

    % Ts annotation
    \draw[as-default,<->,thin,black!70] (0,-0.25) -- (\Ts,-0.25)
      node[midway, below, font=\scriptsize] {$T_s$};
  \end{scope}

  % =========================================================================
  % Parameters
  % =========================================================================
%   \node[below, text width=5cm, align=center, information text] at (0,-\asH/2+1)
%     {$\mathrm{nBits}$ (resolution), $f_s = 1/T_s$\\
%      $\theta_\mathrm{range}$ (angular range)};
}

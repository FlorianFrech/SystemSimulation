% =============================================================================
% Setpoint Component
% =============================================================================
\newcommand{\SetPointComponent}{
\def\bw{8.0}         % component width
\def\bh{4.5}         % component height
\def\gW{6.8}         % mini-graph width
\def\gH{2.0}         % mini-graph height

% Configurable signal parameters (for the visual cue)
\def\setOffset{1.00} % B
\def\setAmp{0.50}    % A
\def\setFreq{0.32}   % f
\def\setPhase{1.05}  % phase [rad]

  \tikzset{setpoint-default/.style={line cap=round, line join=round}}

  \coordinate (C) at (0,0);
  \draw[thick, fill=white!95!black] ($(C)+(-\bw/2,-\bh/2)$) rectangle ($(C)+(\bw/2,\bh/2)$);
  \node[font=\large\bfseries] at ($(C)+(0,\bh/2-0.55)$) {Setpoint};

  % Mini-graph: configurable sinusoidal reference trajectory
  \begin{scope}[shift={(-\gW/2,-\gH/2+0.1)}]
    \draw[setpoint-default,->,thick] (0,0) -- (\gW+0.1,0) node[right] {$t$};
    \draw[setpoint-default,->,thick] (0,0) -- (0,\gH+0.1) node[right] {$\theta_\mathrm{ref}(t)$};

    \draw [help lines, dotted, step=0.5] (0,0) grid (6.9,2.4);

    % Derived signal values
    \pgfmathsetmacro{\setOmega}{2*pi*\setFreq}
    \pgfmathsetmacro{\yStart}{\setOffset + \setAmp*sin(deg(\setPhase))}
    \pgfmathsetmacro{\tPhase}{(pi-\setPhase)/\setOmega}
    \pgfmathsetmacro{\tPeak}{(pi/2-\setPhase)/\setOmega + 2*pi/\setOmega}

    % Offset cue (B)
    \draw[setpoint-default,green!70!black,dashed] (0,\setOffset) -- (\gW,\setOffset);
    \node[font=\scriptsize,anchor=west,green!70!black] at (4.6,\setOffset+0.18) {$B$};

    % Reference signal
    \draw[setpoint-default,angleColor, thick, domain=0:\gW, samples=200, smooth]
      plot (\x,{\setOffset + \setAmp*sin(deg(\setOmega*\x + \setPhase))});

    % Phase cue
    \draw[setpoint-default,densely dashed,red!70] (\tPhase,0) -- (\tPhase,\gH-0.25);
    \draw[setpoint-default,<->,red!85,thin] (0,0.18) -- (\tPhase,0.18);
    \node[font=\scriptsize, red!90] at (2*\tPhase,0.2)
      {phase shift};

    % Amplitude cue (A)
    \pgfmathsetmacro{\yPeak}{\setOffset+\setAmp}
    \draw[setpoint-default,<->,orange!90,thin] (\tPeak,\setOffset) -- (\tPeak,\yPeak);
    \node[font=\scriptsize,orange!90,anchor=south] at (\tPeak+0.12,\setOffset-0.65*\setAmp) {$A$};

    % Initial phase marker
    \fill[angleColor] (0,\yStart) circle(0.025);
    \draw[setpoint-default,densely dashed,black!55] (0,0) -- (0,\yStart);
  \end{scope}

  % Configurable function parameters
  \node[below, text width=7.1cm, align=center, information text] at ($(C)+(0,-1.35)$) {
    $\theta_\mathrm{ref}(t)=B + A\sin\!\left(2\pi f\,t + \phi\right)$};
}

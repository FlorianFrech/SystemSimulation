\tikzset{
  component/.style={
    rectangle,
    draw=blue!50!black,
    fill=blue!12,
    rounded corners=2pt,
    minimum width=20mm,
    minimum height=20mm,
    inner sep=0pt
  },
  port/.style={
    rectangle,
    draw=black,
    fill=white,
    minimum width=7.5mm,
    minimum height=7.5mm,
    inner sep=0pt,
    font=\bfseries\small
  },
  inactiveport/.style={
    port,
    fill=gray!10,
    draw=gray!65,
    dashed,
    text=gray!70!black
  },
  connection/.style={
    thick,
    ->,
    >=Latex
  },
  directfeedthrough/.style={
    ->,
    thick,
    dashed,
    red!75!black
  }
}

\newcommand{\arrowlegend}[1]{%
  \begin{scope}[shift={#1}]
    \draw[->, dashed, thick, red!75!black] (0,0) -- (0.7,0);
    \node[anchor=west, font=\small] at (0.75,0) {direct feedthrough};
    \draw[->, thick, black] (3.8,0) -- (4.5,0);
    \node[anchor=west, font=\small] at (4.55,0) {connection};
  \end{scope}%
}
